## 2. 全局状态管理

### 2.1 useAuthStore（认证状态）

```typescript
interface AuthStore {
  token: string | null;
  user: User | null;
  login: (username: string, password: string) => Promise<void>;
  logout: () => void;
  initialize: () => void;  // 从 localStorage 恢复登录态
}

interface User {
  username: string;
  nickname: string;
}
```

- `token`：登录后存入 localStorage + Zustand
- `login()`：调 `POST /api/v1/auth/login`，成功后写入 `token` 和 `user`，跳转 `/dashboard`
- `logout()`：清除 localStorage 和 Zustand，跳转 `/login`
- `initialize()`：App 启动时检查 localStorage，有 Token 则恢复状态
- axios 拦截器：从 Zustand 获取 Token，注入 `Authorization: Bearer xxx`

### 2.2 错误码统一处理

```typescript
// request.ts - axios 响应拦截器
response.interceptors.use(
  (res) => {
    if (res.data.code !== 0) {
      message.error(res.data.message);
      return Promise.reject(res.data);
    }
    return res.data;  // 只返回 data 字段
  },
  (err) => {
    message.error('网络异常，请稍后重试');
    return Promise.reject(err);
  }
);
```

### 5.2 核心 Hook：useGoalParser

```typescript
// hooks/useGoalParser.ts
interface UseGoalParserReturn {
  loading: boolean;
  parseResult: GoalParseVO | null;
  error: string | null;
  parse: (rawInput: string) => Promise<void>;
  reset: () => void;
}

function useGoalParser(): UseGoalParserReturn {
  const [loading, setLoading] = useState(false);
  const [parseResult, setParseResult] = useState<GoalParseVO | null>(null);
  const [error, setError] = useState<string | null>(null);

  const parse = async (rawInput: string) => {
    setLoading(true);
    setError(null);
    try {
      const res = await goalApi.parse({ rawInput });
      setParseResult(res.data);
      if (res.data.needsClarification) {
        // 需要追问 → 自动跳转澄清页
        window.location.href = `/goals/${res.data.goalId}/clarify`;
      }
    } catch (e) {
      setError('解析失败，请重试');
    } finally {
      setLoading(false);
    }
  };

  const reset = () => { setParseResult(null); setError(null); };
  return { loading, parseResult, error, parse, reset };
}
```

### 6.3 发送回复实现

```typescript
const onSend = async (reply: string) => {
  if (!reply.trim() || loading) return;
  setSending(true);

  // 乐观更新：添加用户消息
  setMessages(prev => [...prev, { role: 'user', content: reply }]);

  try {
    const res = await goalApi.reply(goalId, { reply });

    if (!res.data.needsClarification) {
      message.success('信息已完整，即将进入确认页');
      setTimeout(() => navigate(`/goals/${goalId}/confirm`, { replace: true }), 500);
      return;
    }

    setMessages(prev => [...prev, {
      role: 'agent',
      content: res.data.questions?.join('\n') || '请补充更多信息',
      questions: res.data.questions,
    }]);
    setCurrentRound(res.data.currentRound);

    if (res.data.currentRound >= MAX_CLARIFY_ROUNDS) {
      message.warning('已达到最大追问轮次，将以当前信息生成排期');
      setTimeout(() => navigate(`/goals/${goalId}/confirm`, { replace: true }), 1000);
    }
  } catch {
    setMessages(prev => prev.slice(0, -1)); // 失败回滚
  } finally {
    setSending(false);
  }
};
```

```typescript
function useTaskList() {
  const [tasks, setTasks] = useState<Task[]>([]);
  const [total, setTotal] = useState(0);
  const [loading, setLoading] = useState(false);
  const [filters, setFilters] = useState({
    page: 1, pageSize: 10,
    status: undefined as string | undefined,
    priority: undefined as string | undefined,
    search: undefined as string | undefined,
  });

  const fetchTasks = useCallback(async () => {
    setLoading(true);
    try {
      const res = await taskApi.list({ status: filters.status, priority: filters.priority });
      let data = res.data || [];
      if (filters.search) data = data.filter(t => t.title.includes(filters.search!));
      setTotal(data.length);
      const start = (filters.page - 1) * filters.pageSize;
      setTasks(data.slice(start, start + filters.pageSize));
    } finally {
      setLoading(false);
    }
  }, [filters]);

  useEffect(() => { fetchTasks(); }, [fetchTasks]);

  const updateStatus = async (id: number, status: string) => {
    await taskApi.updateStatus(id, { status });
    fetchTasks();
  };

  const removeTask = async (id: number) => {
    await taskApi.remove(id);
    fetchTasks();
  };

  return {
    tasks, total, loading, filters, setFilters,
    refresh: fetchTasks, updateStatus, removeTask,
    changePage: (p: number) => setFilters(prev => ({ ...prev, page: p })),
  };
}
```

```typescript
// utils/request.ts
import axios from 'axios';

const instance = axios.create({
  baseURL: 'http://localhost:8080',
  timeout: 30000,
  headers: { 'Content-Type': 'application/json' },
});

instance.interceptors.request.use((config) => {
  const token = useAuthStore.getState().token;
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});

instance.interceptors.response.use(
  (res) => {
    if (res.data.code !== 0) {
      message.error(res.data.message || '请求失败');
      return Promise.reject(res.data);
    }
    return res.data;
  },
  (err) => {
    if (err.response?.status === 403) {
      useAuthStore.getState().logout();
      window.location.href = '/login';
    }
    message.error('网络异常，请稍后重试');
    return Promise.reject(err);
  }
);

export default instance;
```
